mp_cat_political = {

	mp_enable_ideology_tab = {
		icon = generic_civil_support
		cost = 0
		priority = 100

		visible = {
			NOT = { has_country_flag = mp_tab_ideology }
			OR = {
				has_tech = democracy_production_research
				has_tech = communism_production_research
				has_tech = fascism_production_research
				has_tech = neutrality_production_research
			}
		}

		complete_effect = {
			set_country_flag = mp_tab_ideology
		}

		ai_will_do = { factor = 0 }
	}

	mp_disable_ideology_tab = {
		icon = generic_civil_support
		cost = 0
		priority = 90

		visible = {
			has_country_flag = mp_tab_ideology
		}

		complete_effect = {
			clr_country_flag = mp_tab_ideology
		}

		ai_will_do = { factor = 0 }
	}


	ideology_research_decision = {
		icon = generic_civil_support
		cost = 0
		priority = 25
		fire_only_once = yes 

		available = {
			NOT = { has_tech = democracy_production_research }
			custom_trigger_tooltip = {
				tooltip = ideology_research_decision_trigger_tt
				OR = {
					has_country_flag = cheat_mode
					check_variable = {
						research_point_var > 49
					}
				}
			}
		}

		complete_effect = {
			if = {
				limit = {
					has_country_flag = normal_mode
					NOT = { has_country_flag = cheat_mode }
				}
				add_to_variable = {
					var = research_point_var
					value = -50
				}
			}
			set_technology = {
				democracy_production_research = 1
				communism_production_research = 1
				neutrality_production_research = 1
				fascism_production_research = 1
			}
		}

		ai_will_do = { factor = 0 }
	}
}
