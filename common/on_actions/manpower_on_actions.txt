on_actions = {

	on_weekly = { # man_in_warehouse_product
		effect = {
			if = {
				limit = {
					check_variable = {
						var:num_equipment@man_in_warehouse_product > 0
					}
				}

				set_variable = {
					var = man_product
					value = var:num_equipment@man_in_warehouse_product
				}
				set_variable = {
					var = man_product_negative
					value = var:num_equipment@man_in_warehouse_product
				}
				multiply_variable = {
					var = man_product_negative
					value = -1
				}

				add_equipment_to_stockpile = {
					type = man_in_warehouse_product
					amount = man_product_negative
				}

				add_manpower = man_product

				set_variable = {
					var = man_product
					value = 0
				}
				set_variable = {
					var = man_product_negative
					value = 0
				}
			}
		}
	}
